项目概述

本系统是一个基于大语言模型(LLM)的A股量化选股策略自动生成与进化系统。核心思想：

  1. LLM生成策略：利用GPT-4/Claude根据市场环境自动生成选股策略
  2. 竞技场验证：多个策略在模拟环境中竞争，优胜劣汰
  3. 达尔文进化：精英策略变异、杂交，持续进化
  4. 知识库增强：RAG系统注入金融学术知识

class TushareDataProvider(DataProvider):
    def __init__(self):
        super().__init__()
        import tushare as ts
        ts.set_token(config.data.tushare_token)
        self.pro = ts.pro_api()
    
    def _safe_call(self, func, max_retries: int = 3, **kwargs):
        for attempt in range(max_retries):
            try:
                result = func(**kwargs)
                if result is not None and not result.empty:
                    return result
                logger.warning(f"API返回空数据 (attempt {attempt+1})")
            except Exception as e:
                logger.warning(f"API调用失败 (attempt {attempt+1}): {e}")
                if attempt < max_retries - 1:
                    time.sleep(2 ** attempt)
        return None
    
    def fetch_stock_basic(self):
        data = self._safe_call(self.pro.stock_basic, 
            exchange='', list_status='L',
            fields='ts_code,symbol,name,area,industry,market,list_date,is_hs')
        if data is not None:
            data = data[data['ts_code'].str.endswith('.SH') | 
                       data['ts_code'].str.endswith('.SZ')]
        return data
    
    def fetch_daily(self, ts_code: str, start_date: str, end_date: str):
        return self._safe_call(self.pro.daily, ts_code=ts_code,
            start_date=start_date, end_date=end_date)
    
    def fetch_daily_batch(self, trade_date: str):
        return self._safe_call(self.pro.daily, trade_date=trade_date)
    
    def fetch_income(self, ts_code: str, start_date: str, end_date: str):
        return self._safe_call(self.pro.income, ts_code=ts_code,
            start_date=start_date, end_date=end_date)
    
    def fetch_balance_sheet(self, ts_code: str, start_date: str, end_date: str):
        return self._safe_call(self.pro.balancesheet, ts_code=ts_code,
            start_date=start_date, end_date=end_date)
    
    def fetch_moneyflow(self, ts_code: str, start_date: str, end_date: str):
        return self._safe_call(self.pro.moneyflow, ts_code=ts_code,
            start_date=start_date, end_date=end_date)

        self.failed_steps = {}
    
    async def daily_update(self):
        today = date.today()
        yesterday = today - timedelta(days=1)
        
        # 获取行情
        df = self.provider.fetch_daily_batch(yesterday.strftime('%Y%m%d'))
        if df is not None:
            self._save_daily_prices(df)
        
        # 更新因子
        await self._update_factors(yesterday)
        
        # 更新缓存

    async def backfill_history(self, years: int = 3):
        end_date = date.today()
        start_date = end_date - timedelta(days=years * 365)
        
        trade_dates = self._get_trade_dates(start_date, end_date)
        logger.info(f"回填 {len(trade_dates)} 个交易日数据...")
        
        for i, trade_date in enumerate(trade_dates):
            if i % 20 == 0:
                logger.info(f"进度: {i}/{len(trade_dates)}")
            
            df = self.provider.fetch_daily_batch(trade_date.strftime('%Y%m%d'))
            if df is not None:
                self._save_daily_prices(df)
            
            if i % 50 == 0:
                time.sleep(1)  # 限流
    
    def _save_daily_prices(self, df: pd.DataFrame):

class BaseFactor(ABC):
    name: str = ""
    category: str = ""
    description: str = ""
    dependencies: List[str] = []
    lookback_days: int = 252
    
    @abstractmethod
    def calculate(self, data: pd.DataFrame) -> pd.Series:
        pass
    
    def validate(self, result: pd.Series) -> bool:
        if result.isna().all():
            return False
        if (result == 0).all():
            return False
        return True
    
    def winsorize(self, series: pd.Series, limits: tuple = (0.01, 0.99)) -> pd.Series:
        lower = series.quantile(limits[0])
        upper = series.quantile(limits[1])
        return series.clip(lower, upper)
    
    def standardize(self, series: pd.Series) -> pd.Series:
        mean = series.mean()
        std = series.std()
        if std == 0:
            return pd.Series(0, index=series.index)
        return (series - mean) / std

class MAFactor(BaseFactor):
    name = "ma"
    description = "均线因子"
    
    def __init__(self, window: int = 20):
        self.window = window
        self.name = f"ma_{window}"
    
    def calculate(self, data: pd.DataFrame) -> pd.Series:
        return data['close'].rolling(window=self.window).mean()

class RSIFactor(BaseFactor):
    name = "rsi"
    description = "RSI相对强弱指标"
    
    def __init__(self, window: int = 14):
        self.window = window
    
    def calculate(self, data: pd.DataFrame) -> pd.Series:
        delta = data['close'].diff()
        gain = delta.where(delta > 0, 0)
        loss = -delta.where(delta < 0, 0)
        avg_gain = gain.rolling(self.window).mean()
        avg_loss = loss.rolling(self.window).mean()
        rs = avg_gain / avg_loss
        return 100 - (100 / (1 + rs))

class MACDFactor(BaseFactor):
    name = "macd"
    description = "MACD指标"
    
    def __init__(self, fast: int = 12, slow: int = 26, signal: int = 9):
        self.fast = fast
        self.slow = slow
        self.signal = signal
    
    def calculate(self, data: pd.DataFrame) -> pd.DataFrame:
        ema_fast = data['close'].ewm(span=self.fast).mean()
        ema_slow = data['close'].ewm(span=self.slow).mean()
        dif = ema_fast - ema_slow
        dea = dif.ewm(span=self.signal).mean()
        hist = 2 * (dif - dea)
        return pd.DataFrame({'dif': dif, 'dea': dea, 'hist': hist})

class MomentumFactor(BaseFactor):
    name = "momentum"
    description = "动量因子：N日收益率"
    
    def __init__(self, window: int = 20):
        self.window = window
        self.name = f"ret_{window}d"
    
    def calculate(self, data: pd.DataFrame) -> pd.Series:
        return data['close'].pct_change(periods=self.window)

class VolatilityFactor(BaseFactor):
    name = "volatility"
    description = "波动率因子"
    
    def __init__(self, window: int = 20):
        self.window = window
    
    def calculate(self, data: pd.DataFrame) -> pd.Series:
        returns = data['close'].pct_change()
        return returns.rolling(window=self.window).std() * np.sqrt(252)

class MaxDrawdownFactor(BaseFactor):
    name = "max_drawdown"
    description = "最大回撤因子"
    
    def __init__(self, window: int = 60):
        self.window = window
    
    def calculate(self, data: pd.DataFrame) -> pd.Series:
        cummax = data['close'].rolling(self.window).max()
        drawdown = (data['close'] - cummax) / cummax
        return drawdown.rolling(self.window).min()

class VectorStoreManager:
    COLLECTIONS = {
        "factor_research": "因子研究知识库",
        "strategy_patterns": "策略模式知识库",
        "market_regimes": "市场环境知识库",
        "academic_papers": "学术论文摘要库",
        "trading_rules": "交易规则与纪律",
        "risk_management": "风险管理知识",
        "strategy_history": "历史策略存档",
    }
    
    def __init__(self, persist_path=None):
        self.client = chromadb.PersistentClient(path=str(persist_path))
        self.embedding_fn = SentenceTransformerEmbeddingFunction(
            model_name="BAAI/bge-large-zh-v1.5",
            normalize_embeddings=True
        )
        self.collections = {}
        self._init_collections()
    
    def query(self, query_text, collection_names=None, n_results=5, 
              min_similarity=0.0):
        results = {}
        for name in (collection_names or self.collections.keys()):
            collection = self.collections[name]
            query_result = collection.query(
                query_texts=[query_text],
                n_results=n_results,
                include=["documents", "metadatas", "distances"]
            )
            formatted = []
            for i in range(len(query_result['ids'][0])):
                distance = query_result['distances'][0][i]
                similarity = 1 - distance
                if similarity >= min_similarity:
                    formatted.append({
                        'document': query_result['documents'][0][i],
                        'similarity': round(similarity, 4)
                    })
            if formatted:
                results[name] = formatted
        return results
    
    def query_combined(self, query_text, n_results=10, min_similarity=0.3):
        """合并查询结果，返回格式化文本用于LLM prompt"""
        all_results = self.query(query_text, n_results=n_results, 
                                min_similarity=min_similarity)
        if not all_results:
            return "（未找到相关知识）"
        parts = []
        for name, docs in all_results.items():
            desc = self.COLLECTIONS.get(name, name)
            parts.append(f"\n### {desc}")
            for i, doc in enumerate(docs[:3], 1):
                parts.append(f"{i}. [相似度:{doc['similarity']:.2f}] {doc['document'][:500]}")
        return "\n".join(parts)

class StrategyCompiler:
    def __init__(self):
        self.factor_registry = self._load_factor_registry()
        self.operator_funcs = {
            ConditionOperator.GT: lambda x, v: x > v,
            ConditionOperator.LT: lambda x, v: x < v,
            ConditionOperator.GTE: lambda x, v: x >= v,
            ConditionOperator.LTE: lambda x, v: x <= v,
            ConditionOperator.EQ: lambda x, v: x == v,
            ConditionOperator.BETWEEN: lambda x, v: (x >= v[0]) & (x <= v[1]),
        }
    
    def compile(self, strategy_dict):
        # 1. 创建Strategy对象
        strategy = Strategy.from_dict(strategy_dict)
        # 2. 验证
        valid, errors = strategy.validate()
        if not valid:
            return False, f"策略验证失败: {'; '.join(errors)}", None, None
        # 3. 验证因子
        valid, msg = self._validate_factors(strategy)
        if not valid:
            return False, msg, None, None
        # 4. 生成执行函数
        strategy_func = self._generate_function(strategy)
        return True, "编译成功", strategy_func, strategy
    
    def _generate_function(self, strategy):
        def select_stocks(data: pd.DataFrame) -> pd.DataFrame:
            if data is None or len(data) == 0:
                return pd.DataFrame()
            df = data.copy()
            # 应用筛选条件
            mask = pd.Series(True, index=df.index)
            for cond in strategy.conditions:
                factor = cond.factor
                if factor not in df.columns:
                    continue
                factor_values = df[factor].copy()
                if cond.operator in self.operator_funcs:
                    mask &= self.operator_funcs[cond.operator](factor_values, cond.value)
            filtered = df[mask].copy()
            if len(filtered) == 0:
                return pd.DataFrame()
            # 打分排序
            if strategy.ranking_factors:
                filtered['_score'] = 0.0
                for rf in strategy.ranking_factors:
                    if rf.factor not in filtered.columns:
                        continue
                    values = filtered[rf.factor]
                    zscore = (values - values.mean()) / (values.std() + 1e-10)
                    if rf.direction == "ascending":
                        zscore = -zscore
                    filtered['_score'] += zscore * rf.weight
                filtered = filtered.sort_values('_score', ascending=False).head(strategy.max_position)
            return filtered
        return select_stocks

class QuickValidator:
    def __init__(self):
        self.min_sharpe = 0.3
        self.min_win_rate = 0.45
        self.history_years = 3
        self.max_conditions = 8
    
    def validate(self, strategy_func, strategy_dict, verbose=True):
        start_time = time.time()
        report = ValidationReport(strategy_name=strategy_dict.get('name', 'unknown'))
        
        # 第1关：逻辑一致性检查
        logic_result = self._check_logic_consistency(strategy_dict)
        if not logic_result['passed']:
            report.warnings.append(f"逻辑问题: {logic_result['issue']}")
            return report
        
        # 第2关：因子有效性检查
        factor_result = self._check_factor_validity(strategy_dict)
        if not factor_result['passed']:
            report.warnings.append(f"因子问题: {factor_result['issue']}")
            return report
        
        # 第3关：历史数据快速扫描
        history_result = self._quick_history_scan(strategy_func, strategy_dict)
        report.metrics = history_result.get('metrics', {})
        if not history_result['passed']:
            report.warnings.append(f"历史验证: {history_result['reason']}")
            return report
        
        # 第4关：统计显著性检验
        returns = history_result.get('returns', np.array([]))
        stat_result = self._statistical_tests(returns)
        if not stat_result['passed']:
            report.warnings.append(f"统计问题: {stat_result['reason']}")
            return report
        
        report.passed = True
        report.validation_time = time.time() - start_time
        return report
    
    def _check_logic_consistency(self, strategy_dict):
        """检查条件矛盾、数量、止损止盈合理性"""
        conditions = strategy_dict.get('conditions', [])
        issues = []
        # 检查重复因子
        factors = [c['factor'] for c in conditions]
        if len(factors) != len(set(factors)):
            issues.append("存在重复因子条件")
        # 检查条件数量
        if len(conditions) > self.max_conditions:
            issues.append(f"条件过多({len(conditions)}>{self.max_conditions})")
        # 检查止损止盈
        stop_loss = strategy_dict.get('stop_loss', -0.08)
        stop_profit = strategy_dict.get('stop_profit', 0.30)
        if abs(stop_loss) > stop_profit:
            issues.append(f"盈亏比不合理: 止损{stop_loss:.0%} > 止盈{stop_profit:.0%}")
        return {'passed': len(issues)==0, 'issue': '; '.join(issues) if issues else ''}
    
    def _statistical_tests(self, returns):
        if len(returns) < 8:
            return {'passed': False, 'reason': f'样本量不足: {len(returns)}'}
        t_stat, p_value = stats.ttest_1samp(returns, 0)
        if p_value > 0.15:
            return {'passed': False, 'reason': f'收益不显著: p={p_value:.3f}', 
                   'p_value': p_value}
        return {'passed': True, 'p_value': round(p_value, 4)}

class LiveStrategy:
    def __init__(self, strategy_id, strategy_func, strategy_meta, 
                 initial_capital=100000.0, commission_rate=0.0003, slippage=0.001):
        self.strategy_id = strategy_id
        self.strategy_func = strategy_func
        self.meta = strategy_meta
        self.name = strategy_meta.get('name', strategy_id)
        self.cash = initial_capital
        self.initial_capital = initial_capital
        self.total_value = initial_capital
        self.positions: Dict[str, Position] = {}
        self.trades: List[Trade] = []
        self.nav_history: List[Dict] = []
        self.stop_loss = strategy_meta.get('stop_loss', -0.08)
        self.stop_profit = strategy_meta.get('stop_profit', 0.30)
        self.max_position = strategy_meta.get('max_position', 10)
        self.commission_rate = commission_rate
        self.slippage = slippage
        self.status = StrategyStatus.ACTIVE
    
    def update_daily(self, market_data, trade_date, prices):
        self._update_positions(prices)
        stop_out = self._check_stop_loss_take_profit(prices, trade_date)
        if self._should_rebalance(trade_date):
            self._rebalance(market_data, prices, trade_date)
        self._update_total_value(prices)
        self.nav_history.append({
            'date': trade_date, 'nav': self.total_value,
            'daily_return': self._calc_daily_return()
        })
    
    def _rebalance(self, market_data, prices, trade_date):
        new_picks = self.strategy_func(market_data)
        if new_picks is None or len(new_picks) == 0:
            return "未选出股票"
        new_codes = set(new_picks.index.tolist())
        current_codes = set(self.positions.keys())
        # 卖出不在新名单中的
        for code in current_codes - new_codes:
            self._sell(code, prices.get(code, 0), trade_date, "调仓卖出")
        # 买入新股
        n_buy = len(new_codes)
        target_weight = min(1.0/n_buy, self.meta.get('single_stock_weight', 0.1))
        for code in new_codes - current_codes:
            price = prices.get(code, 0)
            if price > 0:
                shares = int(self.total_value * target_weight / price / 100) * 100
                if shares > 0 and self.cash >= shares * price * 1.01:
                    self._buy(code, price, shares, trade_date, "新买入")
    
    def _buy(self, code, price, shares, trade_date, reason=""):
        buy_price = price * (1 + self.slippage)
        cost = shares * buy_price
        commission = cost * self.commission_rate
        total_cost = cost + commission
        if total_cost > self.cash * 0.95:
            return
        self.cash -= total_cost
        if code in self.positions:
            old = self.positions[code]
            total_shares = old.shares + shares
            avg_cost = (old.shares * old.avg_cost + cost) / total_shares
            self.positions[code] = Position(ts_code=code, shares=total_shares, 
                                            avg_cost=avg_cost, current_price=buy_price)
        else:
            self.positions[code] = Position(ts_code=code, shares=shares,
                                            avg_cost=buy_price, current_price=buy_price)
    
    def get_stats(self):
        if len(self.nav_history) < 2:
            return {'total_value': self.total_value, 'positions': len(self.positions)}
        navs = [h['nav'] for h in self.nav_history]
        returns = [navs[i]/navs[i-1]-1 for i in range(1, len(navs))]
        returns = np.array(returns)
        total_return = self.total_value / self.initial_capital - 1
        sharpe = returns.mean()/returns.std()*np.sqrt(252) if returns.std()>0 else 0
        cum = np.cumprod(1+returns)
        max_dd = float((cum - np.maximum.accumulate(cum)).min() / np.maximum.accumulate(cum).max())
        win_rate = (returns > 0).mean()
        return {
            'total_return': round(total_return, 4), 'sharpe': round(sharpe, 4),
            'max_drawdown': round(max_dd, 4), 'win_rate': round(win_rate, 4),
            'total_value': round(self.total_value, 2), 'positions': len(self.positions)
        }

class EvolutionEngine:
    def __init__(self, generator, compiler, validator, arena, knowledge_retriever):
        self.generator = generator
        self.compiler = compiler
        self.validator = validator
        self.arena = arena
        self.knowledge = knowledge_retriever
        self.elimination_rate = 0.20
        self.elite_count = 3
        self.mutants_per_elite = 2
        self.new_per_cycle = 3
        self.evolution_cycle = 0
    
    async def evolve(self, market_context=None):
        self.evolution_cycle += 1
        logger.info(f"🧬 进化周期 #{self.evolution_cycle}")
        
        # Phase 1: 评估
        leaderboard = self.arena.get_leaderboard()
        
        # Phase 2: 淘汰末位
        n_eliminate = max(1, int(len(leaderboard) * self.elimination_rate))
        worst = leaderboard.tail(n_eliminate)
        for _, row in worst.iterrows():
            self.arena.remove_strategy(row['strategy_id'])
        
        # Phase 3: 精英识别
        elites = []
        for _, row in leaderboard.head(self.elite_count).iterrows():
            sid = row['strategy_id']
            if sid in self.arena.strategies:
                elites.append({
                    'id': sid, 'name': row['name'],
                    'strategy': self.arena.strategies[sid].meta,
                    'performance': row.to_dict()
                })
        
        # Phase 4: 变异
        mutants = []
        for elite in elites:
            for _ in range(self.mutants_per_elite):
                mutant = self.generator.mutate_strategy(elite['strategy'], elite['performance'])
                if mutant:
                    mutant['parent_strategy_id'] = elite['id']
                    mutant['generation'] = elite['strategy'].get('generation', 0) + 1
                    mutants.append(mutant)
        
        # Phase 5: 杂交
        hybrids = []
        for i in range(len(elites)):
            for j in range(i+1, len(elites)):
                hybrid = self.generator.crossover_strategies(
                    elites[i]['strategy'], elites[j]['strategy'],
                    elites[i]['performance'], elites[j]['performance']
                )
                if hybrid:
                    hybrids.append(hybrid)
        
        # Phase 6: 新生
        new_strategies = self.generator.generate_new_strategies(
            market_context or {}, n_strategies=self.new_per_cycle
        )
        
        # Phase 7: 验证与上线
        all_new = mutants + hybrids + new_strategies
        added = 0
        for strategy_dict in all_new:
            success, msg, func, obj = self.compiler.compile(strategy_dict)
            if not success: continue
            report = self.validator.validate(func, strategy_dict, verbose=False)
            if not report.passed: continue
            if self.arena.add_strategy(obj.id, func, strategy_dict):
                added += 1
        
        return {
            'cycle': self.evolution_cycle,
            'eliminated': n_eliminate,
            'mutated': len(mutants),
            'crossover': len(hybrids),
            'new': len(new_strategies),
            'added': added,
            'arena_size': len(self.arena.strategies)
        }

        memory = psutil.virtual_memory()
        metrics.memory_percent = memory.percent
        metrics.memory_used_gb = memory.used / (1024**3)
        disk = psutil.disk_usage('/')
        metrics.disk_percent = disk.percent
        if self.arena:
            metrics.active_strategies = len(self.arena.strategies)
        return metrics
    
    def check_strategy_alerts(self):
        alerts = []
        for sid, strategy in self.arena.strategies.items():
            stats = strategy.get_stats()
            max_dd = stats.get('max_drawdown', 0)
            if max_dd < -0.20:
                alerts.append(StrategyAlert(
                    strategy_id=sid, strategy_name=strategy.name,
                    alert_type='drawdown', severity='critical' if max_dd < -0.30 else 'warning',
                    message=f"最大回撤 {max_dd:.1%}",
                    current_value=max_dd, threshold=-0.20
                ))
        return alerts
    
    def health_check(self):
        checks = {'status': 'healthy', 'checks': {}}
        try:
            with get_db_session() as session:
                session.execute(text("SELECT 1"))
            checks['checks']['database'] = {'status': 'ok'}
        except Exception as e:
            checks['checks']['database'] = {'status': 'error', 'message': str(e)}
            checks['status'] = 'unhealthy'
        return checks

class AlertManager:
    def __init__(self):
        self.dingtalk_webhook = config.alert.dingtalk_webhook
        self.email_to = config.alert.email_alert
        self._alert_cache = {}
    
    def send_strategy_alert(self, alert):
        dedup_key = f"{alert.strategy_id}_{alert.alert_type}"
        if dedup_key in self._alert_cache:
            if (datetime.now() - self._alert_cache[dedup_key]).seconds < 1800:
                return True
        
        message = f"""
🚨 **策略告警** [{alert.severity.upper()}]
**策略**: {alert.strategy_name}
**类型**: {alert.alert_type}
**详情**: {alert.message}
**当前值**: {alert.current_value}
**阈值**: {alert.threshold}
"""
        sent = False
        if self.dingtalk_webhook:
            sent = self._send_dingtalk(message)
        if self.email_to and alert.severity == 'critical':
            sent = self._send_email(f"[AI量化] {alert.severity}: {alert.strategy_name}", message)
        
        if sent:
            self._alert_cache[dedup_key] = datetime.now()
        return sent
    
    def _send_dingtalk(self, message):
        payload = {"msgtype": "markdown", "markdown": {"title": "AI量化告警", "text": message}}
        response = requests.post(self.dingtalk_webhook, json=payload, timeout=10)
        return response.status_code == 200 and response.json().get('errcode') == 0

    async def run(self):

        # 每日: 数据更新 15:30, 模拟交易 16:00, 告警 16:30, 报告 17:00

        self.scheduler.add_job(jobs.weekly_evolution, 'cron', day_of_week='sat', hour=10)
